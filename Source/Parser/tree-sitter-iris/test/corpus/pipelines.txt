module Pipelines;

function run(list: Array_slice::<Complex>, weights: Array_slice::<Float32>, threshold: Float32) -> ()
{
    pipeline process(input: Complex, weight: Float32)
    {
        map scale(input, weight) => v0;
        map v0->real => v1;
        map v1 * weight => v2;
        filter v2 > threshold;
    }

    for v in process(list, weights)
    {
        print(v);
    }
}
